# Læg denne fil i testmappen ved siden af index.html, og kald den
# præcis  .htaccess  — med punktum foran og uden .txt til sidst.
#
# Den gør tre ting: beder søgemaskiner om at holde sig væk, spærrer for
# at mappens indhold kan listes, og sørger for at browseren aldrig viser
# en gammel udgave af siden.
#
# De to første skal SLETTES, når beregneren flytter til sin rigtige
# adresse og skal kunne findes. Cache-reglen bør blive: uden den kan en
# kunde, der har set siden før, få en forældet udgave serveret i dagevis
# efter en rettelse — og I ville ikke kunne se det ske.

<IfModule mod_headers.c>
  Header set X-Robots-Tag "noindex, nofollow, noarchive"

  # Selve siden må aldrig komme fra cachen. Den er ét dokument med det
  # hele i, så der er ikke andet at hente — og den vejer under 300 KB.
  <FilesMatch "\.html$">
    Header set Cache-Control "no-cache, no-store, must-revalidate"
    Header set Pragma "no-cache"
    Header set Expires "0"
  </FilesMatch>
</IfModule>

Options -Indexes
